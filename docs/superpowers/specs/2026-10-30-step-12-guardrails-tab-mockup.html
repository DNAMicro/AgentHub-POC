    // rule: ca-drivers-license  v1 (draft)
export const triggers = [
  /\b[A-Z]\d{7}\b/i,                // cheap pre-check
];

const CONTEXT = /\b(licen[cs]e|DL|driver'?s|ID)\b/i;
const SHAPE = /\b[A-Z]\d{7}\b/gi;

export function detect(text) {
  const found = [];
  for (const m of text.matchAll(SHAPE)) {
    const near = text.slice(Math.max(0, m.index - 60),
                            m.index + 68);
    found.push({
      kind: 'ca_drivers_license',
      start: m.index, end: m.index + m[0].length,
      confidence: CONTEXT.test(near) ? 'high' : 'low',
    });
  }
  return found;   // positions only, never the value
}
